function hexToRgbString(hex) {
  const normalized = hex.replace("#", "");
  const full = normalized.length === 3 ? normalized.split("").map((char) => char + char).join("") : normalized;
  const int = Number.parseInt(full, 16);
  const r = (int >> 16) & 255;
  const g = (int >> 8) & 255;
  const b = int & 255;
  return `${r} ${g} ${b}`;
}

"use client";

import { useEffect, useRef, useState } from "react";
import { useLocale } from "@/components/LocaleProvider";
import { personal } from "@/data/resumeContent";
import themeConfig from "@/data/themeConfig.json";
import { gsap } from "@/lib/gsap";
import { usePrefersReducedMotion } from "@/hooks/usePrefersReducedMotion";

function getInitials(name) {
  if (!name) return "CV";
  const parts = name.trim().split(/\s+/).filter(Boolean);
  if (parts.length === 1) return parts[0].slice(0, 1).toUpperCase();
  const first = parts[0].slice(0, 1).toUpperCase();
  const last = parts[parts.length - 1].slice(0, 1).toUpperCase();
  return `${first}/${last}`;
}

export default function LoaderOverlay() {
  const { locale } = useLocale();
  const [visible, setVisible] = useState(true);
  const rootRef = useRef(null);
  const reducedMotion = usePrefersReducedMotion();
  const loaderText = locale === "en" ? "m.a" : themeConfig.loader?.textValue || getInitials(personal.name);

  useEffect(() => {
    if (!visible) return undefined;

    const body = document.body;
    const html = document.documentElement;
    const previousOverflow = body.style.overflow;
    let didReveal = false;
    window.__cv2LoaderDone = false;
    html.classList.add("cv2-loading");
    body.style.overflow = "hidden";

    const revealMain = () => {
      if (didReveal) return;
      didReveal = true;
      window.__cv2LoaderDone = true;
      window.dispatchEvent(new Event("cv2:loader-done"));
      html.classList.remove("cv2-loading");
    };

    const markDone = () => {
      body.style.overflow = previousOverflow;
      setVisible(false);
    };

    if (reducedMotion || !rootRef.current) {
      revealMain();
      markDone();
      return undefined;
    }

    const ctx = gsap.context(() => {
      const loader = rootRef.current;
      const initialsNode = loader.querySelector(".loader-initials");
      const leftPanel = loader.querySelector(".loader-panel-left");
      const rightPanel = loader.querySelector(".loader-panel-right");

      gsap.set(initialsNode, { autoAlpha: 0, scale: 1, y: 10, filter: "blur(0px)" });
      gsap.to(initialsNode, { autoAlpha: 1, y: 0, duration: 0.7, ease: "power2.out" });

      const loadPromise = new Promise((resolve) => {
        if (document.readyState === "complete") {
          resolve(true);
          return;
        }
        window.addEventListener("load", () => resolve(true), { once: true });
      });

      Promise.all([loadPromise, new Promise((resolve) => setTimeout(resolve, 1100))]).then(() => {
        const tl = gsap.timeline({
          defaults: { ease: "power3.inOut" },
          onComplete: markDone
        });

        tl.to(initialsNode, {
          scale: 2.4,
          filter: "blur(16px)",
          autoAlpha: 0,
          duration: 0.8
        })
          .call(revealMain)
          .to(
            leftPanel,
            {
              xPercent: -104,
              duration: 1.05
            },
            "-=0.08"
          )
          .to(
            rightPanel,
            {
              xPercent: 104,
              duration: 1.05
            },
            "<"
          )
          .to(loader, { autoAlpha: 0, duration: 0.28 }, "-=0.14");
      });
    }, rootRef);

    return () => {
      revealMain();
      html.classList.remove("cv2-loading");
      body.style.overflow = previousOverflow;
      ctx.revert();
    };
  }, [visible, reducedMotion]);

  if (!visible) return null;

  return (
    <div ref={rootRef} className="fixed inset-0 z-[140]">
      <div className="absolute inset-0 flex">
        <div className="loader-panel-left h-full w-1/2" style={{ backgroundColor: "var(--loader-bg)" }} />
        <div className="loader-panel-right h-full w-1/2" style={{ backgroundColor: "var(--loader-bg)" }} />
      </div>

      <div className="absolute inset-0 flex items-center justify-center">
        <span
          className="loader-initials display-font text-[clamp(3.8rem,15vw,13rem)] font-extrabold leading-none tracking-[0.08em]"
          style={{ color: "var(--loader-text)" }}
        >
          {loaderText}
        </span>
      </div>
    </div>
  );
}

"use client";

import { createContext, useCallback, useContext, useEffect, useMemo, useState } from "react";
import * as persianContent from "@/data/resumeContent";
import { englishContent, englishUi } from "@/data/englishContent";

const LocaleContext = createContext(null);
const englishResume = { ...persianContent, ...englishContent };

export function LocaleProvider({ children }) {
  const [locale, setLocale] = useState("fa");

  useEffect(() => {
    const savedLocale = window.localStorage.getItem("portfolio-locale");
    if (savedLocale === "en" || savedLocale === "fa") setLocale(savedLocale);
  }, []);

  useEffect(() => {
    const root = document.documentElement;
    root.lang = locale;
    root.dir = locale === "fa" ? "rtl" : "ltr";
    root.dataset.locale = locale;
    window.localStorage.setItem("portfolio-locale", locale);

    const title = locale === "fa"
      ? `${persianContent.personal.name} | ${persianContent.siteMeta.titleSuffix}`
      : `${englishContent.personal.name} | ${englishContent.siteMeta.titleSuffix}`;
    document.title = title;
    const titleTimeout = window.setTimeout(() => {
      document.title = title;
    }, 250);

    const description = document.querySelector('meta[name="description"]');
    const localizedDescription = locale === "fa" ? persianContent.siteMeta.description : englishContent.siteMeta.description;
    if (description) {
      description.setAttribute("content", localizedDescription);
    }

    ["og:title", "twitter:title"].forEach((name) => {
      document.querySelector(`meta[property="${name}"], meta[name="${name}"]`)?.setAttribute("content", title);
    });
    ["og:description", "twitter:description"].forEach((name) => {
      document.querySelector(`meta[property="${name}"], meta[name="${name}"]`)?.setAttribute("content", localizedDescription);
    });
    const openGraphLocale = document.querySelector('meta[property="og:locale"]');
    if (openGraphLocale) openGraphLocale.setAttribute("content", locale === "fa" ? "fa_IR" : "en_US");
    return () => window.clearTimeout(titleTimeout);
  }, [locale]);

  const t = useCallback((text) => (
    locale === "en" ? englishUi[text] || text : text
  ), [locale]);
  const content = locale === "en" ? englishResume : persianContent;
  const value = useMemo(() => ({ locale, setLocale, t, content }), [content, locale, t]);

  return <LocaleContext.Provider value={value}>{children}</LocaleContext.Provider>;
}

export function useLocale() {
  const context = useContext(LocaleContext);
  if (!context) throw new Error("useLocale must be used within LocaleProvider");
  return context;
}

export function useResumeContent() {
  return useLocale().content;
}

"use client";

import { useEffect } from "react";
import { gsap } from "@/lib/gsap";
import { usePrefersReducedMotion } from "@/hooks/usePrefersReducedMotion";

export default function SectionGlideController() {
  const reducedMotion = usePrefersReducedMotion();

  useEffect(() => {
    if (reducedMotion) return undefined;

    const sections = gsap.utils.toArray("[data-section]");
    const triggers = [];

    sections.forEach((section) => {
      if (!(section instanceof HTMLElement)) return;
      if (section.id === "journey") return;

      const target =
        section.querySelector("[data-glide-target]") ||
        section.querySelector(".section-shell") ||
        section.firstElementChild;

      if (!target || !(target instanceof HTMLElement)) return;

      gsap.set(target, { willChange: "transform" });
      const tween = gsap.fromTo(
        target,
        { y: 26 },
        {
          y: -20,
          ease: "none",
          scrollTrigger: {
            trigger: section,
            start: "top bottom",
            end: "bottom top",
            scrub: 1.35
          }
        }
      );

      triggers.push(() => {
        tween.scrollTrigger?.kill();
        tween.kill();
        gsap.set(target, { clearProps: "willChange" });
      });
    });

    return () => triggers.forEach((kill) => kill());
  }, [reducedMotion]);

  return null;
}

function formatTime(value, locale) {
  const safe = Number.isFinite(value) ? Math.max(0, value) : 0;
  const minutes = Math.floor(safe / 60);
  const seconds = Math.floor(safe % 60);
  return `${formatLocaleNumber(minutes, locale, { minimumIntegerDigits: 2 })}:${formatLocaleNumber(seconds, locale, { minimumIntegerDigits: 2 })}`;
}

function PodcastPlayer({ src, title }) {
  const { locale, t } = useLocale();
  const audioRef = useRef(null);
  const [isPlaying, setIsPlaying] = useState(false);
  const [currentTime, setCurrentTime] = useState(0);
  const [duration, setDuration] = useState(0);

  useEffect(() => {
    const audio = audioRef.current;
    if (!audio) return undefined;

    const onLoaded = () => setDuration(audio.duration || 0);
    const onTime = () => setCurrentTime(audio.currentTime || 0);
    const onEnded = () => setIsPlaying(false);

    audio.addEventListener("loadedmetadata", onLoaded);
    audio.addEventListener("timeupdate", onTime);
    audio.addEventListener("ended", onEnded);

    return () => {
      audio.removeEventListener("loadedmetadata", onLoaded);
      audio.removeEventListener("timeupdate", onTime);
      audio.removeEventListener("ended", onEnded);
    };
  }, []);

  const toggle = async () => {
    const audio = audioRef.current;
    if (!audio) return;

    if (audio.paused) {
      try {
        await audio.play();
        setIsPlaying(true);
      } catch {
        setIsPlaying(false);
      }
      return;
    }

    audio.pause();
    setIsPlaying(false);
  };

  const progressPercent = duration > 0 ? (currentTime / duration) * 100 : 0;

  return (
    <div className="mt-4 rounded-2xl border border-white/15 bg-black/25 p-3 md:p-4">
      <audio ref={audioRef} preload="none" className="hidden">
        <source src={src} type="audio/mpeg" />
      </audio>

      <div className="flex items-center gap-3">
        <button
          type="button"
          onClick={toggle}
          aria-label={`${t(isPlaying ? "مکث پادکست" : "پخش پادکست")} ${title}`}
          className="group inline-flex h-10 w-10 items-center justify-center rounded-full border border-white/25 bg-white/[0.06] transition hover:border-neon-cyan/65 hover:bg-neon-cyan/10"
        >
          {isPlaying ? (
            <span className="flex gap-1">
              <span className="h-3 w-1 bg-neon-cyan" />
              <span className="h-3 w-1 bg-neon-cyan" />
            </span>
          ) : (
            <span className="ml-0.5 h-0 w-0 border-b-[7px] border-l-[11px] border-t-[7px] border-b-transparent border-l-neon-cyan border-t-transparent" />
          )}
        </button>

        <div className="min-w-0 flex-1">
          <input
            type="range"
            min={0}
            max={duration || 0}
            value={currentTime}
            aria-label={`${t("موقعیت پخش پادکست")} ${title}`}
            onChange={(event) => {
              const next = Number(event.target.value || 0);
              const audio = audioRef.current;
              if (!audio) return;
              audio.currentTime = next;
              setCurrentTime(next);
            }}
            className="h-1 w-full cursor-pointer appearance-none rounded-full bg-white/15 accent-cyan-400"
            style={{
              background: `linear-gradient(90deg, rgba(55,244,255,0.95) ${progressPercent}%, rgba(255,255,255,0.16) ${progressPercent}%)`
            }}
          />
          <div className="mt-2 flex items-center justify-between">
            <span className="mono-font text-[10px] uppercase tracking-[0.18em] text-silver/65">{formatTime(currentTime, locale)}</span>
            <span className="mono-font text-[10px] uppercase tracking-[0.18em] text-silver/65">{formatTime(duration, locale)}</span>
          </div>
        </div>
      </div>
    </div>
  );
}

const fs = require('fs');
const path = require('path');

const ROOT_DIR = __dirname;
const OUTPUT_FILE = path.join(ROOT_DIR, 'project_dump.txt');

// فایل‌ها و پوشه‌هایی که نباید خوانده شوند
const IGNORE_DIRS = ['node_modules', '.next', '.git', 'out', 'dist'];
const IGNORE_EXTS = ['.png', '.jpg', '.jpeg', '.webp', '.ico', '.woff', '.woff2', '.ttf', '.mp3', '.mp4'];
const ALLOWED_EXTS = ['.js', '.jsx', '.ts', '.tsx', '.css', '.json', '.md'];

let outputContent = '';

function readDir(dir) {
    const files = fs.readdirSync(dir);

    for (const file of files) {
        const fullPath = path.join(dir, file);
        const stat = fs.statSync(fullPath);

        if (stat.isDirectory()) {
            if (IGNORE_DIRS.includes(file)) continue;
            readDir(fullPath);
        } else {
            const ext = path.extname(file).toLowerCase();
            
            // فقط فایل‌های متنی و کد را بخوان
            if (ALLOWED_EXTS.includes(ext) && !IGNORE_EXTS.includes(ext)) {
                // اگر در پوشه public بود و عکس بود، فقط اسمش را لیست کن
                if (dir.includes('public') && IGNORE_EXTS.includes(ext)) continue;

                const relativePath = path.relative(ROOT_DIR, fullPath);
                const content = fs.readFileSync(fullPath, 'utf8');

                outputContent += `\n${'='.repeat(80)}\n`;
                outputContent += `FILE: ${relativePath}\n`;
                outputContent += `${'='.repeat(80)}\n\n`;
                outputContent += content;
                outputContent += '\n\n';
            }
        }
    }
}

console.log('⏳ در حال استخراج ساختار پروژه...');
readDir(ROOT_DIR);
fs.writeFileSync(OUTPUT_FILE, outputContent, 'utf8');
console.log(`✅ تمام شد! فایل خروجی در مسیر زیر ساخته شد:\n${OUTPUT_FILE}`);

"use client";

import { useEffect, useState } from "react";

export function usePrefersReducedMotion() {
  const [prefersReducedMotion, setPrefersReducedMotion] = useState(false);

  useEffect(() => {
    const mediaQuery = window.matchMedia("(prefers-reduced-motion: reduce)");
    const onChange = () => setPrefersReducedMotion(mediaQuery.matches);

    onChange();
    mediaQuery.addEventListener("change", onChange);
    return () => mediaQuery.removeEventListener("change", onChange);
  }, []);

  return prefersReducedMotion;
}

const basePath = process.env.NEXT_PUBLIC_BASE_PATH || "";

export function withBasePath(assetPath) {
  if (!assetPath) return assetPath;
  if (/^https?:\/\//i.test(assetPath)) return assetPath;
  const normalized = assetPath.startsWith("/") ? assetPath : `/${assetPath}`;
  return `${basePath}${normalized}`;
}

function getNumberLocale(locale) {
  return locale === "fa" ? "fa-IR-u-nu-arabext" : "en-US-u-nu-latn";
}

export function formatLocaleNumber(value, locale, options = {}) {
  return new Intl.NumberFormat(getNumberLocale(locale), options).format(value);
}

export function formatLocaleYear(year, locale) {
  if (locale !== "fa") {
    return formatLocaleNumber(year, locale, { useGrouping: false });
  }

  const date = new Date(Date.UTC(Number(year), 6, 1));
  return new Intl.DateTimeFormat("fa-IR-u-ca-persian-nu-arabext", {
    year: "numeric",
    timeZone: "UTC"
  }).format(date);
}

export function formatLocaleYearLabel(value, locale) {
  if (value == null || value === "") return "";

  const normalized = String(value).replace(/[۰-۹٠-٩]/g, (digit) => {
    const code = digit.charCodeAt(0);
    return String(code >= 0x06f0 ? code - 0x06f0 : code - 0x0660);
  });

  return normalized.replace(/\b(?:18|19|20|21)\d{2}\b/g, (year) => formatLocaleYear(year, locale));
}

export function formatLocaleDate(value, locale) {
  const date = value instanceof Date ? value : new Date(value);
  const dateLocale = locale === "fa"
    ? "fa-IR-u-ca-persian-nu-arabext"
    : "en-US-u-ca-gregory-nu-latn";

  return new Intl.DateTimeFormat(dateLocale, { dateStyle: "medium", timeZone: "UTC" }).format(date);
}

# محمد آیت | توسعه‌دهنده فول‌استک وب

وب‌سایت رزومه و نمونه‌کارهای من؛ فضایی برای معرفی مسیر حرفه‌ای، مهارت‌ها و پروژه‌هایی که ساخته‌ام.

## مشاهده وب‌سایت

[bitwizarder.github.io](https://bitwizarder.github.io)

## امکانات

- طراحی فارسی و راست‌چین، سازگار با موبایل و دسکتاپ
- پیمایش بخش‌ها با نقشه‌ی مسیر و حرکت‌های نرم
- معرفی پروژه‌ها با پیوند مستقیم به مخزن‌های گیت‌هاب
- نمایش آمار و مخزن‌های منتخب گیت‌هاب
- دریافت اطلاعات عمومی گیت‌هاب هنگام ساخت سایت
- انتشار خودکار روی GitHub Pages با GitHub Actions

## فناوری‌ها

- Next.js و React
- JavaScript
- Tailwind CSS
- GSAP و ScrollTrigger

## اجرای محلی

```bash
npm install
npm run dev
```

سپس [http://localhost:3000](http://localhost:3000) را باز کنید.

برای ساخت نسخه‌ی قابل انتشار:

```bash
npm run snapshot:github
npm run build
```

خروجی ایستا در پوشه‌ی `out/` ساخته می‌شود.

## ویرایش محتوا

اطلاعات رزومه، متن‌ها و روشن/خاموش‌کردن بخش‌ها در [data/resumeContent.js](./data/resumeContent.js) قرار دارد. برای تغییر رنگ‌ها و تنظیمات ظاهری، [data/themeConfig.json](./data/themeConfig.json) را ویرایش کنید.

شناسه‌ی کاربری گیت‌هاب در `githubProfile.username` تنظیم شده است. اسکریپت `npm run snapshot:github` اطلاعات عمومی پروفایل و مخزن‌ها را در `data/githubSnapshot.json` به‌روزرسانی می‌کند.

## انتشار در GitHub Pages

مخزن برای انتشار خودکار با GitHub Actions آماده شده است:

1. در تنظیمات مخزن، بخش **Pages** را باز کنید.
2. منبع انتشار را روی **GitHub Actions** قرار دهید.
3. تغییرات را به شاخه‌ی `main` بفرستید.

گردش‌کار [deploy-pages.yml](./.github/workflows/deploy-pages.yml) اطلاعات گیت‌هاب را دریافت می‌کند، سایت را می‌سازد و منتشر می‌کند. تنظیم مسیر خروجی نیز بین مخزن کاربری `username.github.io` و مخزن‌های پروژه تفاوت قائل می‌شود.
